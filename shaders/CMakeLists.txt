# shaders/CMakeLists.txt — vendor JS-built-in GLSL into a static library.
#
# Each .glsl file under this directory is embedded as a `const char*` in
# the generated `builtin_shaders.cc`. The lookup is exposed via
# `declgl::builtin_shader_source(name, kind)` from `builtin_shaders.h`.
#
# Naming convention: <program>.<vert|frag>.glsl
#
#   triangle.vert.glsl  → ("triangle", VERT)
#   triangle.frag.glsl  → ("triangle", FRAG)

set(_SHADER_FILES
    triangle.vert.glsl
    triangle.frag.glsl
    rect.vert.glsl
    rect.frag.glsl
    circle.vert.glsl
    circle.frag.glsl
    roundedRect.frag.glsl
    texture.vert.glsl
    texture.frag.glsl
    textureCentered.vert.glsl
    textureCentered.frag.glsl
    textureCroppedCentered.vert.glsl
)

set(_GEN_HDR "${CMAKE_CURRENT_BINARY_DIR}/builtin_shaders.h")
set(_GEN_SRC "${CMAKE_CURRENT_BINARY_DIR}/builtin_shaders.cc")

# Public header. Stable; doesn't need to be regenerated, but keeping it
# alongside the .cc is convenient.
file(WRITE  "${_GEN_HDR}"
"// Generated by shaders/CMakeLists.txt — do not edit.
#pragma once
#include <string_view>
namespace declgl {
enum class ShaderKind { VERT, FRAG };
// Returns the embedded GLSL source for the given (program, kind), or an
// empty string_view if no such builtin is vendored.
std::string_view builtin_shader_source(std::string_view program,
                                       ShaderKind kind);
}  // namespace declgl
")

# The .cc body is generated by a CMake script run at build time so it
# automatically picks up shader edits.
set(_GEN_SCRIPT "${CMAKE_CURRENT_BINARY_DIR}/gen_builtin_shaders.cmake")
file(WRITE "${_GEN_SCRIPT}"
"# Generated by shaders/CMakeLists.txt — do not edit.
file(WRITE  \"${_GEN_SRC}\"
\"// Generated by shaders/CMakeLists.txt — do not edit.
#include \\\"builtin_shaders.h\\\"
#include <string>
#include <unordered_map>
namespace declgl {
namespace {
struct Key { std::string program; ShaderKind kind; bool operator==(const Key& o) const { return program == o.program && kind == o.kind; } };
struct KeyHash { size_t operator()(const Key& k) const noexcept { return std::hash<std::string>{}(k.program) ^ (size_t)k.kind; } };
const std::unordered_map<Key, std::string_view, KeyHash>& table() {
    static const std::unordered_map<Key, std::string_view, KeyHash> t = {\")
")

# Append one map entry per shader file. Each entry uses C++ raw string
# literals so we don't have to escape anything.
foreach(_f IN LISTS _SHADER_FILES)
    string(REPLACE ".glsl" "" _stem "${_f}")
    # _stem is e.g. "triangle.vert"
    string(REPLACE "." ";" _parts "${_stem}")
    list(GET _parts 0 _prog)
    list(GET _parts 1 _kindlc)
    if(_kindlc STREQUAL "vert")
        set(_kind "VERT")
    elseif(_kindlc STREQUAL "frag")
        set(_kind "FRAG")
    else()
        message(FATAL_ERROR "Bad shader kind '${_kindlc}' for ${_f}")
    endif()
    file(APPEND "${_GEN_SCRIPT}"
"
file(READ \"${CMAKE_CURRENT_SOURCE_DIR}/${_f}\" _src)
file(APPEND \"${_GEN_SRC}\"
    \"        { Key{\\\"${_prog}\\\", ShaderKind::${_kind}}, R\\\"GLSL(\")
file(APPEND \"${_GEN_SRC}\" \"\${_src}\")
file(APPEND \"${_GEN_SRC}\" \")GLSL\\\" },\\n\")
")
endforeach()

file(APPEND "${_GEN_SCRIPT}"
"
file(APPEND \"${_GEN_SRC}\"
\"    };
    return t;
}
}  // namespace
std::string_view builtin_shader_source(std::string_view program, ShaderKind kind) {
    auto& t = table();
    auto it = t.find(Key{std::string(program), kind});
    return it == t.end() ? std::string_view{} : it->second;
}
}  // namespace declgl
\")
")

add_custom_command(
    OUTPUT  ${_GEN_SRC}
    COMMAND ${CMAKE_COMMAND} -P ${_GEN_SCRIPT}
    DEPENDS ${_SHADER_FILES} ${_GEN_SCRIPT}
    WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
    COMMENT "Embedding builtin GLSL shaders -> builtin_shaders.cc"
    VERBATIM
)

add_library(declgl_shaders STATIC ${_GEN_SRC} ${_GEN_HDR})
set_target_properties(declgl_shaders PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(declgl_shaders PUBLIC ${CMAKE_CURRENT_BINARY_DIR})
